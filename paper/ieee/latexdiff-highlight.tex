%% latexdiff preamble for the IEEE Access "Highlighted PDF". Additions are
%% set in blue and deletions are omitted, so the highlighted copy paginates
%% like the clean copy with every changed passage marked. Color-only markup
%% is used deliberately: both background highlighters available under
%% pdfTeX (soul's \hl and ulem's mark-over) fail on this class's captions,
%% run-in headings, and the citation-dense prose, and a highlighted copy
%% that does not compile marks nothing. Passed to latexdiff --preamble by
%% scripts/pack-ieee-submission.sh.
\RequirePackage{color}
\providecommand{\DIFaddtex}[1]{{\protect\color{blue}#1}}
\providecommand{\DIFdeltex}[1]{}
\providecommand{\DIFadd}[1]{\DIFaddtex{#1}}
\providecommand{\DIFdel}[1]{\DIFdeltex{#1}}
\providecommand{\DIFaddbegin}{}
\providecommand{\DIFaddend}{}
\providecommand{\DIFdelbegin}{}
\providecommand{\DIFdelend}{}
\providecommand{\DIFmodbegin}{}
\providecommand{\DIFmodend}{}

\providecommand{\DIFaddFL}[1]{\DIFadd{#1}}
\providecommand{\DIFdelFL}[1]{\DIFdel{#1}}
\providecommand{\DIFaddbeginFL}{}
\providecommand{\DIFaddendFL}{}
\providecommand{\DIFdelbeginFL}{}
\providecommand{\DIFdelendFL}{}
