%% latexdiff preamble for the IEEE Access "Highlighted PDF": additions are
%% highlighted in yellow (the editorial office's stated preference) and
%% deletions are dropped, so the highlighted copy paginates like the clean
%% copy with the changed passages marked. The highlighter is ulem's
%% mark-over construction (a yellow rule laid under every character), which
%% breaks across lines and survives the citations, monospace paths, and
%% table cells that soul's \hl cannot reconstruct. Passed to latexdiff
%% --preamble by scripts/pack-ieee-submission.sh.
\RequirePackage[normalem]{ulem}
\RequirePackage{color}
\newcommand{\DIFhighlight}{\bgroup\markoverwith{\textcolor{yellow}{\rule[-.55ex]{2pt}{2.5ex}}}\ULon}
\providecommand{\DIFaddtex}[1]{\DIFhighlight{#1}}
\providecommand{\DIFdeltex}[1]{}
\providecommand{\DIFadd}[1]{\DIFaddtex{#1}}
\providecommand{\DIFdel}[1]{\DIFdeltex{#1}}
\providecommand{\DIFaddbegin}{}
\providecommand{\DIFaddend}{}
\providecommand{\DIFdelbegin}{}
\providecommand{\DIFdelend}{}
\providecommand{\DIFmodbegin}{}
\providecommand{\DIFmodend}{}
%DIF FLOATSAFE PREAMBLE
%% Captions are moving arguments in which ulem recurses under this class;
%% additions inside them are marked by color instead of the yellow rule.
\providecommand{\DIFaddFL}[1]{{\protect\color{blue}#1}}
\providecommand{\DIFdelFL}[1]{\DIFdel{#1}}
\providecommand{\DIFaddbeginFL}{}
\providecommand{\DIFaddendFL}{}
\providecommand{\DIFdelbeginFL}{}
\providecommand{\DIFdelendFL}{}
